\Cref{alg:tool-search} gives the ranking of the tool search (\cref{sec:tool-search}).
The name weighs most, because data servers name tools by function, such as \texttt{vn\_quote\_history}.
Vietnamese phrases such as \enquote{lịch sử giá} (price history) are translated to find tools described in English, and partial matches need words of three letters or more.

\begin{algorithm}[H]
  \caption{Ranking of deferred tools for a search query}\label{alg:tool-search}
  \begin{algorithmic}[1]
    \Require query $q$; deferred tools $D$, each with name, search hint, description and parameter names
    \Ensure tools of $D$ ordered by relevance to $q$
    \If{$q$ has the form \texttt{select:}$a,b,\dots$} \State \Return the tools named $a,b,\dots$ \EndIf
    \State $W \gets$ words of $q$ with Vietnamese diacritics removed; words marked \texttt{+} are required
    \State add to $W$ the English terms of every Vietnamese market phrase found in $q$, longest phrase first
    \For{each tool $d \in D$}
      \State $s_d \gets \sum_{w \in W}$ ( 12 if $w$ is a word of the name, else 6 if it is part of one;
      \Statex \hspace{\algorithmicindent}\hspace{\algorithmicindent} $+4$ if in the search hint; $+2$ if in the description; $+2$ if a parameter name )
      \If{a required word scores 0 for $d$} \State discard $d$ \EndIf
    \EndFor
    \State \Return at most 5 tools (by default) with $s_d>0$ and $s_d \ge \tfrac{1}{2}\max_{e\in D} s_e$, by decreasing $s_d$, then by name
  \end{algorithmic}
\end{algorithm}
